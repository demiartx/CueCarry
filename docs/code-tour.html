      2. Paths and rekordbox "Location" URLs

      What Convert between rekordbox's file://localhost/... URLs and normal paths.

64def loc_to_path(loc: str) -> str:
65    """rekordbox Location URL -> file path with forward slashes.
66
67    Windows:  file://localhost/J:/Music/a.flac  ->  J:/Music/a.flac
68    macOS:    file://localhost/Users/me/a.flac  ->  /Users/me/a.flac
69    """
70    p = loc[len(PREFIX):] if loc.startswith(PREFIX) else loc.replace("file://", "", 1)

95def common_folder(paths: List[str]) -> str:
96    """Deepest folder that contains every path (case-insensitive compare)."""
97    if not paths:
98        return ""
99    split = [norm(posixpath.dirname(norm(p))).split("/") for p in paths]
100    first = split[0]
101    n = len(first)
102    for parts in split[1:]:
103        n = min(n, len(parts))
104        for i in range(n):
105            if parts[i].lower() != first[i].lower():
106                n = i
107                break

111def _volume(path: str) -> str:
112    """Which drive a path is on: 'j:' on Windows, '/volumes/usb' or '/users' on macOS."""
113    parts = norm(path).lower().split("/")
114    if re.match(r"^[a-z]:$", parts[0]):
115        return parts[0]
116    if path.startswith("//") and len(parts) > 2:
117        return "//" + parts[2]                       # network share
118    if len(parts) > 2 and parts[1] == "volumes":
119        return "/volumes/" + parts[2]
120    return "/" + (parts[1] if len(parts) > 1 else "")

123def detect_src_root(paths: List[str]) -> str:
124    """Common folder of the tracks on the drive that holds the most of them.
125
126    Tracks on other drives (for example rekordbox's sampler sounds on C:) are
127    then reported as "outside the source folder" instead of forcing the
128    source folder up to nothing.
129    """
130    if not paths:
131        return ""
132    groups: Dict[str, List[str]] = {}
133    for p in paths:
134        groups.setdefault(_volume(p), []).append(p)

277def make_plan(lib: Library, s: Settings) -> Plan:
278    convertible = [t for t in lib.tracks.values() if t.ext in LOSSLESS_EXTS | PROBE_EXTS]
279    # Detect the source folder from the WHOLE collection, not just the chosen
280    # playlists, so a test run and the full run put files in the same place.
281    src_root = norm(s.src_root) if s.src_root else detect_src_root([t.path for t in convertible])
282    dst_root = norm(s.dst_root) if s.dst_root else src_root + " MP3"
283    if is_under(dst_root, src_root):
284        raise ValueError("The MP3 folder must not be inside the source folder.")
285    out_xml = s.out_xml or posixpath.join(dst_root, "rekordbox_mp3.xml")
286
287    jobs: List[Job] = []
288    keep: List[Track] = []
289    skipped: List[tuple] = []
290    for tid in tracks_in_scope(lib, s):
291        t = lib.tracks.get(tid)
292        if t is None:
293            continue
294        if t.ext in KEEP_EXTS:
295            keep.append(t)
296        elif t.ext in LOSSLESS_EXTS | PROBE_EXTS:

302            skipped.append((t, f"unsupported file type ({t.ext or 'none'})"))

      5. Cue shifting

      What How far to move every cue, loop and grid point for one track.

      Why The MP3 starts later than the source by the encoder delay (section 1). "compensate" removes the header so every player sees that same delay, then shifts by exactly 1105 samples at the MP3's sample rate.

      If it goes wrong cues land early or late by ~25 ms. Run Tools > Make calibration tracks to measure what your rekordbox does.

317def mp3_sample_rate(source_rate: int) -> int:
318    """MP3 tops out at 48 kHz; higher rates are resampled (see encoder.py)."""
319    if source_rate <= 48000:
320        return source_rate
321    return 44100 if source_rate % 44100 == 0 else 48000
322
323
324def cue_shift_seconds(track: Track, s: Settings) -> float:
325    if s.mode == "compensate":
326        rate = mp3_sample_rate(int(float(track.element.get("SampleRate") or 44100)))
327        return ENCODER_DELAY_SAMPLES / rate
328    if s.mode == "offset":
329        return s.offset_ms / 1000.0
330    return 0.0  # align: the audio itself is moved, cues stay put

387def write_xml(lib: Library, plan: Plan, s: Settings,
388              is_alac: Callable[[str], bool]) -> tuple:
389    """Write plan.out_xml. Returns (number of tracks, number of playlists).
390
391    is_alac(path) decides .m4a files that were never converted: ALAC files
392    without an MP3 yet are left out; AAC files are kept as they are.
393    """
394    kbps = int(re.sub(r"\D", "", s.bitrate) or 320)

401        elif j.probe and os.path.exists(j.track.path) and not is_alac(j.track.path):

403            ids.add(j.track.track_id)

405        tracks.append(t.element)
406        ids.add(t.track_id)
407
408    out = ET.Element("DJ_PLAYLISTS", {"Version": "1.0.0"})
409    ET.SubElement(out, "PRODUCT", {"Name": "rekordbox", "Version": "7", "Company": "AlphaTheta"})

411    col.extend(tracks)
412    pls = ET.SubElement(out, "PLAYLISTS")

      7. Reading calibration results

      What Find the four calibration tracks in an exported collection XML and report how far each one's beat grid sits from the FLAC reference.

      Why rekordbox's grid start (TEMPO Inizio) is precise to 1 ms, which beats judging waveforms by eye.

      If it goes wrong "not found" means rekordbox hasn't analyzed the tracks yet, or the collection was exported before they were imported.

437def read_calibration(xml_path: str) -> List[tuple]:
438    """Returns [(title, grid start in s or None, shift vs reference in ms or None)]."""
439    lib = load_library(xml_path)
440    found: Dict[str, Optional[float]] = {}
441    for t in lib.tracks.values():
442        title = t.element.get("Name", "")
443        if title.startswith("CAL ") and t.element.get("Artist") == "Calibration":
444            tempo = t.element.find("TEMPO")
445            found[title] = float(tempo.get("Inizio")) if tempo is not None else None
446    ref = next((v for k, v in found.items() if k.startswith("CAL A")), None)
447    rows = []
448    for title in sorted(found):
449        v = found[title]
450        shift = None
451        if v is not None and ref is not None:
452            d = (v - ref) % 0.5          # beats are 0.5 s apart at 120 BPM
453            if d > 0.25:
454                d -= 0.5
455            shift = round(d * 1000, 1)
456        rows.append((title, v, shift))
457    return rows

      3. Building one ffmpeg command

      What The exact ffmpeg arguments for one track.

      Why - 320 kbps constant bitrate, best encoder quality (-compression_level 0) - ID3v2.3 tags (the version DJ software reads most reliably) - cover art copied when present - tags from the rekordbox XML override the file's own tags - compensate/align modes leave out the LAME header (-write_xing 0) so every player sees the same start delay - align mode trims the delay off the start (atrim) - rates above 48 kHz are resampled, since MP3 stops at 48 kHz

      If it goes wrong the ffmpeg error is shown in the log and saved in the failures file; the first attempt keeps cover art, the second drops it.

152def ffmpeg_commands(ffmpeg: str, job: Job, s: Settings, tmp: str) -> List[List[str]]:
153    attrs = job.track.element.attrib
154    base = [ffmpeg, "-nostdin", "-hide_banner", "-loglevel", "error", "-y", "-i", job.track.path]
155    audio = ["-c:a", "libmp3lame", "-b:a", s.bitrate, "-compression_level", "0",
156             "-map_metadata", "0", "-id3v2_version", "3", "-write_id3v1", "1"]
157    rate = int(float(attrs.get("SampleRate") or 0) or 0)
158    if rate > 48000:
159        audio += ["-ar", str(mp3_sample_rate(rate))]
160    if s.mode == "align":
161        audio += ["-af", f"atrim=start_sample={s.trim_samples},asetpts=PTS-STARTPTS"]
162    if s.mode in ("compensate", "align"):
163        audio += ["-write_xing", "0"]
164    tail = tag_args(attrs) + ["-f", "mp3", tmp]
165    with_art = ["-map", "0:a:0", "-map", "0:v?", "-c:v", "copy", "-disposition:v", "attached_pic"]
166    no_art = ["-map", "0:a:0"]
167    return [base + with_art + audio + tail, base + no_art + audio + tail]

      4. Running the whole plan

      What Encode every job on a pool of threads, report progress after each track, and stop cleanly when asked.

      Why Encoding is the slow part; one ffmpeg per CPU core keeps the machine busy. Each MP3 is written as "name.mp3.part" and renamed when done, so a stopped run never leaves a half-written MP3 behind.

      If it goes wrong failures are collected with their ffmpeg error; running again retries only what is missing.

182class RunResult:
183    done: int = 0
184    converted: int = 0

190ProgressFn = Callable[[int, int, str, str], None]   # (done, total, file name, status)

192
193class Runner:
194    def __init__(self, ffmpeg: str, plan: Plan, s: Settings,
195                 progress: ProgressFn, cancel: Optional[threading.Event] = None):
196        self.ffmpeg, self.plan, self.s, self.progress = ffmpeg, plan, s, progress
197        self.cancel = cancel or threading.Event()
198        self.lock = threading.Lock()
199        self.procs: set = set()
200        self.result = RunResult()
201
202    def stop(self) -> None:
203        """Called from the window's Stop button (another thread)."""
204        self.cancel.set()
205        with self.lock:
206            for p in list(self.procs):
207                try:
208                    p.kill()
209                except OSError:
210                    pass
211
212    def _run_cmd(self, cmd: List[str]) -> tuple:
213        p = subprocess.Popen(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE,
214                             creationflags=NO_WINDOW)
215        with self.lock:
216            self.procs.add(p)
217        _, err = p.communicate()
218        with self.lock:
219            self.procs.discard(p)
220        return p.returncode, err.decode("utf-8", "replace").strip()
221
222    def _one(self, job: Job) -> str:
223        """Encode one track. Any error becomes a "FAIL ..." status for that
224        track instead of stopping the whole run."""
225        try:
226            return self._encode(job)
227        except Exception as e:   # noqa: BLE001 - reported per track
228            return f"FAIL {type(e).__name__}: {e}"
229
230    def _encode(self, job: Job) -> str:
231        if self.cancel.is_set():
232            return "cancelled"
233        if not os.path.exists(job.track.path):
234            return "FAIL source file missing"
235        if job.probe and audio_codec(self.ffmpeg, job.track.path) != "alac":
236            return "kept (AAC, already lossy)"
237        os.makedirs(os.path.dirname(job.dst), exist_ok=True)
238        tmp = job.dst + ".part"
239        err = ""
240        for cmd in ffmpeg_commands(self.ffmpeg, job, self.s, tmp):
241            code, err = self._run_cmd(cmd)
242            if self.cancel.is_set():
243                break
244            if code == 0 and os.path.exists(tmp) and os.path.getsize(tmp) > 0:
245                _retry(os.replace, tmp, job.dst)
246                return "ok"
247        if os.path.exists(tmp):
248            _retry(os.remove, tmp)
249        if self.cancel.is_set():
250            return "cancelled"
251        return "FAIL " + (err.splitlines()[-1] if err else "ffmpeg failed")
252
253    def run(self) -> RunResult:
254        todo = self.plan.todo(self.s.overwrite)
255        total = len(todo)
256
257        def work(job: Job) -> None:
258            status = self._one(job)
259            with self.lock:
260                if status == "cancelled":
261                    return
262                self.result.done += 1
263                if status == "ok":
264                    self.result.converted += 1
265                elif status.startswith("kept"):
266                    self.result.kept_aac += 1
267                else:
268                    self.result.failures.append((job.track.path, status[5:]))
269                done = self.result.done
270            self.progress(done, total, os.path.basename(job.dst), status)
271
272        with cf.ThreadPoolExecutor(max(1, self.s.jobs)) as ex:
273            list(ex.map(work, todo))
274        self.result.cancelled = self.cancel.is_set()
275        return self.result

278def _retry(fn, *args, tries: int = 20, wait: float = 0.25):
279    """Windows can briefly lock a file that was just written (antivirus
280    scanning, the search indexer). Retry a rename or delete for ~5 seconds,
281    like a C++ loop around MoveFileEx that sleeps on ERROR_SHARING_VIOLATION."""
282    for i in range(tries):
283        try:
284            return fn(*args)
285        except PermissionError:
286            if i == tries - 1:
287                raise
288            time.sleep(wait)
